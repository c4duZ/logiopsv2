```cpp
// FocusInfo: the only data the match engine ever sees. Either field MAY be empty
// when the compositor cannot expose it (degradation is encoded as empty strings,
// never as a crash).
struct FocusInfo {
    std::string windowClass;  // X11 WM_CLASS instance/class, or Wayland app_id
    std::string executable;   // basename of the focused process exe, or ""
};

// Abstract watcher. A backend emits onFocusChanged(FocusInfo) whenever the
// foreground app changes. supported() reports whether THIS session/compositor
// can actually provide foreground info (false => documented degradation: per-app
// switching disabled, manual profiles still work).
class FocusWatcher {
public:
    virtual ~FocusWatcher() = default;
    virtual bool supported() const = 0;          // false => degraded mode
    virtual void start() = 0;                     // begin watching
    virtual void stop() = 0;
    // Callback registration (single consumer: the match engine driver in Plan 02).
    std::function<void(const FocusInfo&)> onFocusChanged;
};
```
