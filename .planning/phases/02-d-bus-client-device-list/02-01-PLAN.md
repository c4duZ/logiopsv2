```cpp
// src/logiops-gui/DeviceModel.h (created in Plan 03)
class DeviceModel : public QAbstractListModel {
  Q_OBJECT
public:
  enum Roles {
    PathRole = Qt::UserRole + 1, // QString
    NameRole,                    // QString  (DEV-04)
    ModelRole,                   // QString
    DeviceKindRole,              // int {Mouse=0, Keyboard=1, Unknown=2}
    BatteryPercentRole,          // int 0..100 (DEV-02)
    BatteryKnownRole,            // bool      (false -> "—")
    ChargingRole,                // bool      (DEV-02)
    ConnectionStateRole,         // int {Online=0, Sleeping=1, Offline=2, Unknown=3}
  };
  int rowCount(const QModelIndex& = {}) const override;
  QVariant data(const QModelIndex&, int role) const override;
  QHash<int,QByteArray> roleNames() const override;
public slots:
  // Test driver calls these directly (no live bus needed):
  void onDeviceAdded(const QString& path, const QString& name,
                     quint16 pid, bool active);
  void onDeviceRemoved(const QString& path);
  void onStatusChanged(const QString& path, bool active);
  void onBatteryChanged(const QString& path, int percent, bool charging, bool known);
  void clear(); // CONF-03 reconnect: drop all rows, re-enumerate from scratch
};
```

BatteryStatus HID++ wrapper (Plan 02 will implement; BatteryStatusTest drives the static decoder):
```cpp
// src/logid/backend/hidpp20/features/BatteryStatus.h (created in Plan 02)
namespace logid::backend::hidpp20 {
  class BatteryStatus : public Feature {
  public:
    static constexpr uint16_t ID = FeatureID::BATTERY_STATUS; // 0x1000
    enum Function { GetBatteryLevelStatus = 0 };
    enum Event : uint8_t { BatteryStatusBroadcast = 0 };
    struct Status { uint8_t percentage; bool charging; bool known; };
    // PURE decoder over a hidpp::Report — unit-testable with no device:
    static Status decodeStatus(const hidpp::Report& report);
  };
}
```
